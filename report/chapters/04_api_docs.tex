\section{RESTful APIs}
The server interacts with the client purely through HTTP protocols. The API design adheres strictly to standard RESTful architectures mapping HTTP verbs directly to CRUD operations on underlying resources. Every protected endpoint receives information about the logged-in client through the \texttt{@AuthenticationPrincipal} construct, securely extracting the user identity from the JWT token provided in the HTTP headers.

\section{Detailed Endpoints Description}

\subsection{Authentication Controller}

\vspace{0.3cm}
\noindent \apiPOST\ \texttt{/api/auth/register} \\
\textbf{Access:} Public \\
\textbf{Body:} \texttt{\{ "name": string, "surname": string, "email": string, "password": string, "iban": string? \}} \\
\textbf{Response:} \texttt{\{ "user": UserResponseDTO, "token": string \}} \\
\textbf{Description:} Accepts user registration details, creates a new account, and returns the user profile along with a JWT session token.

\vspace{0.3cm}
\noindent \apiPOST\ \texttt{/api/auth/login} \\
\textbf{Access:} Public \\
\textbf{Body:} \texttt{\{ "email": string, "password": string \}} \\
\textbf{Response:} \texttt{\{ "user": UserResponseDTO, "token": string \}} \\
\textbf{Description:} Authenticates a user based on their credentials and returns their profile with a new JWT session token.

\subsection{User Controller}

\vspace{0.3cm}
\noindent \apiGET\ \texttt{/api/users/me} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Response:} \texttt{UserResponseDTO} \\
\textbf{Description:} Retrieves the profile information of the currently authenticated user.

\vspace{0.3cm}
\noindent \apiPATCH\ \texttt{/api/users/me} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Body:} \texttt{\{ "name": string?, "surname": string?, "iban": string?, "paymentLink": string? \}} \\
\textbf{Response:} \texttt{UserResponseDTO} \\
\textbf{Description:} Updates specific fields of the current user's profile and returns the updated data.

\subsection{Household Controller}

\vspace{0.3cm}
\noindent \apiPOST\ \texttt{/api/households} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Body:} \texttt{\{ "name": string \}} \\
\textbf{Response:} \texttt{HouseholdResponseDTO} \\
\textbf{Description:} Instantiates a new household and automatically assigns the creator as the household administrator.

\vspace{0.3cm}
\noindent \apiPOST\ \texttt{/api/households/members} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Body:} \texttt{\{ "invitationCode": string \}} \\
\textbf{Response:} \texttt{HouseholdResponseDTO} \\
\textbf{Description:} Joins an existing household using a valid invitation code.

\vspace{0.3cm}
\noindent \apiGET\ \texttt{/api/households/me} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Response:} \texttt{HouseholdResponseDTO} \\
\textbf{Description:} Retrieves details about the user's current household.

\vspace{0.3cm}
\noindent \apiGET\ \texttt{/api/households/members} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Response:} \texttt{List<HouseholdMemberResponseDTO>} \\
\textbf{Description:} Retrieves the list of all members currently belonging to the user's household.

\vspace{0.3cm}
\noindent \apiDELETE\ \texttt{/api/households/members/\{memberId\}} \\
\textbf{Access:} @AuthenticationPrincipal (Admin) \\
\textbf{Path Params:} \texttt{memberId} (UUID) \\
\textbf{Response:} \texttt{HouseholdResponseDTO} \\
\textbf{Description:} Removes a specific member from the household. The requester must be an administrator.

\vspace{0.3cm}
\noindent \apiDELETE\ \texttt{/api/households/me} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Response:} \texttt{204 No Content} \\
\textbf{Description:} Allows the current user to voluntarily leave their household.

\vspace{0.3cm}
\noindent \apiGET\ \texttt{/api/households/invitation-code} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Response:} \texttt{HouseholdInvitationCodeResponseDTO} \\
\textbf{Description:} Retrieves the current active invitation code for the household.

\vspace{0.3cm}
\noindent \apiPOST\ \texttt{/api/households/invitation-code/refresh} \\
\textbf{Access:} @AuthenticationPrincipal (Admin) \\
\textbf{Response:} \texttt{HouseholdInvitationCodeResponseDTO} \\
\textbf{Description:} Generates a new unique invitation code, invalidating the previous one.

\subsection{Expense Controller}

\vspace{0.3cm}
\noindent \apiPOST\ \texttt{/api/expenses} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Body:} \texttt{ExpenseCreateRequestDTO} \\
\textbf{Response:} \texttt{ExpenseResponseDTO} \\
\textbf{Description:} Registers a new expense, automatically computing each participant's share based on the selected split type, and generates the resulting debt edges.

\vspace{0.3cm}
\noindent \apiGET\ \texttt{/api/expenses} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Query Params:} \texttt{householdId, userTransactionRole, dateRange, description} \\
\textbf{Response:} \texttt{List<ExpenseResponseDTO>} \\
\textbf{Description:} Retrieves a filtered list of expenses associated with the requesting user's household.

\vspace{0.3cm}
\noindent \apiGET\ \texttt{/api/expenses/overview} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Response:} \texttt{ExpenseOverviewResponseDTO} \\
\textbf{Description:} Returns aggregated expense statistics (total amount and count) for the current calendar month within the household.

\vspace{0.3cm}
\noindent \apiGET\ \texttt{/api/expenses/me} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Response:} \texttt{UserNetOverviewResponseDTO} \\
\textbf{Description:} Returns the net cash flow amount for the requesting user over the current month.

\subsection{Debt Controller}

\vspace{0.3cm}
\noindent \apiGET\ \texttt{/api/debts} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Query Params:} \texttt{userTransactionRole, involvedId} \\
\textbf{Response:} \texttt{List<DebtResponseDTO>} \\
\textbf{Description:} Retrieves a filtered list of the user's debt records, either as a creditor or debtor.

\vspace{0.3cm}
\noindent \apiGET\ \texttt{/api/debts/me} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Response:} \texttt{DebtOverviewResponseDTO} \\
\textbf{Description:} Summarizes the aggregate debt position of the user, showing total amounts owed by and to the user.

\vspace{0.3cm}
\noindent \apiDELETE\ \texttt{/api/debts/\{debtId\}} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Path Params:} \texttt{debtId} (UUID) \\
\textbf{Response:} \texttt{204 No Content} \\
\textbf{Description:} Manually deletes the specified debt record.

\subsection{Settlement Controller}

\vspace{0.3cm}
\noindent \apiPOST\ \texttt{/api/settlements/\{debtId\}} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Path Params:} \texttt{debtId} (UUID) \\
\textbf{Body:} \texttt{\{ "amount": decimal, "description": string? \}} \\
\textbf{Response:} \texttt{SettlementResponseDTO} \\
\textbf{Description:} Creates a settlement transaction for the specified debt, reducing its outstanding balance accordingly.

\vspace{0.3cm}
\noindent \apiGET\ \texttt{/api/settlements} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Query Params:} \texttt{householdId, userTransactionRole, dateRange, description} \\
\textbf{Response:} \texttt{List<SettlementResponseDTO>} \\
\textbf{Description:} Retrieves a filtered list of settlement transactions.

\subsection{Chore Controller}

\vspace{0.3cm}
\noindent \apiPOST\ \texttt{/api/chores} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Body:} \texttt{ChoreCreateRequestDTO} \\
\textbf{Response:} \texttt{ChoreResponseDTO} \\
\textbf{Description:} Defines a new household chore with its description and recurrence parameters.

\vspace{0.3cm}
\noindent \apiPOST\ \texttt{/api/chores/assignments} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Body:} \texttt{ChoreAssignmentCreateRequestDTO} \\
\textbf{Response:} \texttt{ChoreAssignmentResponseDTO} \\
\textbf{Description:} Creates a specific assignment for an existing chore to a selected user.

\vspace{0.3cm}
\noindent \apiGET\ \texttt{/api/chores} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Response:} \texttt{List<ChoreResponseDTO>} \\
\textbf{Description:} Retrieves all base chores defined within the current household.

\vspace{0.3cm}
\noindent \apiGET\ \texttt{/api/chores/assignments} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Query Params:} \texttt{ChoreAssignmentFilterRequestDTO} \\
\textbf{Response:} \texttt{List<ChoreAssignmentResponseDTO>} \\
\textbf{Description:} Retrieves filtered chore assignments based on status, dates, and assignee.

\vspace{0.3cm}
\noindent \apiGET\ \texttt{/api/chores/assignments/me} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Response:} \texttt{AssignmentOverviewDTO} \\
\textbf{Description:} Returns total pending and overdue assignments counts for the requesting user.

\vspace{0.3cm}
\noindent \apiGET\ \texttt{/api/chores/assignments/overview} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Response:} \texttt{AssignmentOverviewDTO} \\
\textbf{Description:} Returns total pending and overdue assignments counts for the entire household.

\vspace{0.3cm}
\noindent \apiPATCH\ \texttt{/api/chores/assignments/\{assignmentId\}/status} \\
\textbf{Access:} @AuthenticationPrincipal \\
\textbf{Path Params:} \texttt{assignmentId} (UUID) \\
\textbf{Body:} \texttt{ChoreStatusUpdateRequestDTO} \\
\textbf{Response:} \texttt{ChoreAssignmentResponseDTO} \\
\textbf{Description:} Updates the completion status of a specific chore assignment.

\vspace{0.3cm}
\noindent \apiDELETE\ \texttt{/api/chores/\{choreId\}} \\
\textbf{Access:} @AuthenticationPrincipal (Admin) \\
\textbf{Path Params:} \texttt{choreId} (UUID) \\
\textbf{Response:} \texttt{204 No Content} \\
\textbf{Description:} Deletes a chore and cascades removal to all its associated assignments.

\vspace{0.3cm}
\noindent \apiDELETE\ \texttt{/api/chores/assignments/\{assignmentId\}} \\
\textbf{Access:} @AuthenticationPrincipal (Admin) \\
\textbf{Path Params:} \texttt{assignmentId} (UUID) \\
\textbf{Response:} \texttt{204 No Content} \\
\textbf{Description:} Deletes a specific chore assignment.
